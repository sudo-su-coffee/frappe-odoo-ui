item_price.py 
import frappe
from frappe.model.document import Document

class ItemPrice(Document):
    def validate(self):
        # Native validation lifecycle
        if self.price_list_rate <= 0:
            frappe.throw("Rate must be strictly positive")
            
        # Auto-calculate margin via doc events
        self.apply_currency_exchange()

    def apply_currency_exchange(self):
        rate = frappe.db.get_single_value(
            "Accounts Settings", "default_currency"
        )
        self.currency = rate
